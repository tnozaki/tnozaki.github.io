## C++ で継承はどのように実装されるのか

C++ で書いた以下のコードは

```c++
class IHasValue {
public:
	virtual int getValue() = NULL;
};
class Foo : IHasValue {
private:
	int foo = 1;
public:
	int getValue() {
		return foo;
	}
};
class Bar : IHasValue {
private:
	int bar = 2;
public:
	int getValue() {
		return bar;
	}
};
class Buzz : public Foo, public Bar {
};
int
main()
{
	Buzz* buzz = new Buzz();
	if (buzz != NULL) {
		std::cout << ((Foo *)buzz)->getValue() << ((Bar *)buzz)->getValue() << std::endl;
		delete buzz;
	}
	return EXIT_SUCCESS;
}
```

以下の C コードと同じになるはずである。

```c
#include <stdio.h>
#include <stdlib.h>

/* メモリアロケータ */
#define operator_new		malloc
#define operator_delete		free

/* IHasValue 純粋仮想関数テーブル */
struct IHasValueVFTBL {
	int (*getValue)(struct Foo*);
};
/* クラス Foo */
struct FooVFTBL;
struct Foo {
	const struct FooVFTBL* vftbl;
	int foo;
};
/* Foo 仮想関数テーブル */
struct FooVFTBL {
	int (*getValue)(struct Foo*);
	void (*Delete)(struct Foo*);
};
static int Foo_getValue(struct Foo*);
void Foo_Delete(struct Foo*);
static struct FooVFTBL Foo_vftbl = {
	&Foo_getValue,
	& Foo_Delete

Foo_Foo(struct Foo* this_)
{
	this_->vftbl = &Foo_vftbl;
	this_->foo = 1;
	return this_;

{
	return this_->foo;
}

Foo_Delete(struct Foo* this_)
{
	operator_delete(this_);
}

	int bar;
};

	void (*Delete)(struct Bar*);

	&Bar_getValue,
	&Bar_Delete

{
	this_->vftbl = &Bar_vftbl;
	this_->bar = 2;
	return this_;

{
	return this_->bar;
}
/* Bar デストラクタ */

	operator_delete(this_);
}

/* クラス Buzz */
struct Buzz {
	struct Foo foo;
	struct Bar bar;
};

Buzz_Buzz(struct Buzz* this_)
{
	Foo_Foo(&this_->foo);
	Bar_Bar(&this_->bar);
	return this_;
}
/* Buzz デストラクタ */
void
Buzz_Delete(struct Buzz* this_)
{
	operator_delete(this_);
}

#define CAST_BUZZ_BAR(this_)	(&(this_)->bar)

int
main()
{
	struct Buzz* buzz;
	int foo, bar;

	buzz = (struct Buzz*)operator_new(sizeof(*buzz));
	if (buzz != NULL) {
		Buzz_Buzz(buzz);
		foo = CAST_BUZZ_FOO(buzz)->vftbl->getValue(CAST_BUZZ_FOO(buzz));
		bar = CAST_BUZZ_BAR(buzz)->vftbl->getValue(CAST_BUZZ_BAR(buzz));
		printf("%d%d\n", foo, bar);
		Buzz_Delete(buzz);
	}
	return EXIT_SUCCESS;
}
```

注目するところは `CAST_BUZZ_*` マクロ。
C++ には多重継承とかいう嫌なものがあるので、キャストを行う時にこのマクロと同様のオフセット調整をコンパイラが頑張ってやらないととならない。

この反省で後発の Java は多重継承を禁止し、純粋仮想関数テーブルだけ interface として独立させたのである。
しかしその後 Java8 でバイナリ互換のため interface にデフォルト実装が持たせられるという仕様変更がなされ、結果として C++ の多重継承と同じトラブルが発生するようになって大草原。

まぁ C なんかでも Unix の `dlopen/dlsym` や `LoadLibrary/GetProcAddress` を使ってプラグイン的なものを実装する時、新しい ABI の本体が古い ABI のプラグインを呼び込むとインタフェース不足するからからシム足したりするけどさ。
N の libc でも俺が弄った時 i18n module に足りない場合は libc 側でシムを足すとかやってる、
[これ](https://cvsweb.netbsd.org/bsdweb.cgi/src/lib/libc/citrus/citrus_ctype.c.diff?r1=1.3;r2=1.4)
とか
[これ](https://codeberg.org/tnozaki/netbsd/commit/2217cf606d985e22d326a66f70bb197c2c32e5b5)
。


このキャスト時のオフセット調整を C++ では Thunking と呼ぶ。

## 閑話 Thunking とは何か

いいですよね The Beatles の Abbey Road B 面のメドレーの中の一曲、ってそれは Sun King や。
元々は think の過去形 thought の赤ちゃん言葉である thunk に現在分詞 ing までくっついたもはや謎の造語である。

プログラミング界隈だと ALGOL で名前呼出や遅延評価において値の計算に使われる計算式のことをさす。うーんこれじゃ何のことかさっぱり判らん。
ギークの言語野がぶっ壊れてるのは東も西も同じやな…

C++ においては遅延評価以外にも、多重継承のために `this` をキャスト先の型に応じてオフセットを調整する必要があるのを上のコードで理解いただけたはずだがこれを Thunking と呼ぶのだ。

つまりは「帳尻合わせ」程度の意味だと思っておけばよろしい。

それ以外でも C/C++ では Thunking という用語は使われる、例えば 32bit なプログラムと 64bit なプログラムで通信する際に ILP32 と LLP64 の違いによって型のサイズ計算の帳尻を合わせる必要があり、これも Thunking と呼ばれる。
Unix の LP64 と違って Windows は LLP64 だから long は 32bit のままなので影響は少ないとはいえ、ポインタは 4byte から 8byte になるからしゃーない。

ちょうど今解析してる CtDPOXY.DLL もユーザ空間から `DeviceIoControl` システムコール経由ででカーネルドライバを叩くプロキシ DLL なので Thunking は無縁ではない。
32bit の CtDPROXY.DLL から 64bit のカーネルドライバを叩くような組み合わせだと、カーネルの 64bit アドレスをユーザー空間の 32bit で扱えるよう「帳尻を合わせる」必要があるからね。

そのため CtDPROXT.DLL は 32bit 版でも

* Binary\Drivers\wdm\common\i386\CtDPROXY.DLL … 32bit 版
* Binary\Drivers\wdm\common\i386\thunk\CtDPROXY.DLL … 32bit Thunking 版

の 2 種類が存在するのだ、今解析してるのは　32bit Thunking 版である。
