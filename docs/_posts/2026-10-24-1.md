## `tlhelp32.h` によるプロセス名の列挙

前々回の復元コードのの中でひとつ気になった個所。

```c++
#include <windows.h>
#include <tchar.h>
#include <tlhelp32.h>

HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
if (snapshot != INVALID_HANDLE_VALUE) {
	PROCESSENTRY32 processEntry;
	BOOL missingPatchMixDSP = TRUE;
	ZeroMemory(&processEntry, sizeof(processEntry));
	processEntry.dwSize = sizeof(processEntry);
	if (Process32First(snapshot, &processEntry) == TRUE) {
		do {
			if (!_tcscmp(processEntry.szExeFile, procname)) {
				missingPatchMixDSP = FALSE;
				break;
			}
		} while (Process32Next(snapshot, &processEntry) == TRUE);
	}
	CloseHandle(snapshot);
...
}
```

通常の Win32 API の命名規約であれば

* `PROCESSENTRY32`, `Process32First`, `Process32Next` はマクロ
* コンパイルオプションで Character Set を Multibyte にした場合は `PROCESSENTRY32A` `Process32FirstA` `Process32NextA` のエイリアス
* 〃 Unicode 〃 `PROCESSENTRY32W` `Process32FirstW` `Process32NextW` 〃

になってるはずである、ところが `tlhelp32.h` の Multibyte API の 末尾には A がつかずそのまんま `PROCESSENTRY32`, `Process32First`, `Process32Next` として実装されてんのよね。

なんで命名規約に沿ってないのかというとおそらくは文字列扱う API には A or W をつけるルールが徹底される前に実装されたからと思われる。
名前に 32 が入ってるから 32bit な 95/NT 4.0 以降に実装されたっぽいけどその時期でもまだそんなんだったっけ？

ただまぁこれでも実用上はなんら問題ないはずである普通のコード書いてれば、ところがちょっと邪悪なプログラミングをしようと考えてしまったぼくは罠に嵌ってしまったのだった。

```c++
#include <windows.h>
#include <tchar.h>
#include <tlhelp32.h>

#define CONCAT(a,b)	a##b

#define RUNNING_PROCESS_PROTO_DECL(suffix) \
typedef PROC(*CONCAT(ConsumeRunningProcess,suffix))(PROC, PVOID, CONCAT(PPROCESSENTRY32,suffix), LSTATUS*); \
LSTATUS CONCAT(RunningProcessProvider,suffix)(PVOID, CONCAT(ConsumeRunningProcess,suffix));

RUNNING_PROCESS_PROTO_DECL(A, PCSTR, PSTR)
RUNNING_PROCESS_PROTO_DECL(W, PCWSTR, PWSTR)

#if defined(UNICODE)
#define ConsumeRunningProcess	ConsumeRunningProcessW
#define RunningProcessProvider	RunningProcessProviderW
#else
#define ConsumeRunningProcess	ConsumeRunningProcessA
#define RunningProcessProvider	RunningProcessProviderA
#endif

#define RUNNING_PROCESS_IMPL_DEF(suffix) \
LSTATUS \
CONCAT(RunningProcessProvider,suffix)(PVOID param, CONCAT(ConsumeRunningProcess,suffix) consume) \
{ \
	LSTATUS ret = ERROR_SUCCESS; \
	HANDLE snapshot; \
	CONCAT(PROCESSENTRY32,suffix) processEntry; \
 \
	/* param may be NULL */ \
	/* consume may be NULL*/ \
 \
	if (consume != NULL) { \
		snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0); \
		if (snapshot == INVALID_HANDLE_VALUE) { \
			ret = GetLastError(); \
		} else { \
			if (CONCAT(Process32First,suffix)(snapshot, &processEntry) == TRUE) { \
				do { \
					consume = (CONCAT(ConsumeRunningProcess,suffix))(*consume)((PROC)consume, param, &processEntry, &ret); \
					if (consume == NULL) \
						break; \
				} while (CONCAT(Process32Next,suffix)(snapshot, &processEntry) == TRUE); \
			} \
			CloseHandle(snapshot); \
		} \
	} \
	return ERROR_SUCCESS; \
}

RUNNING_PROCESS_IMPL_DEF(A)
RUNNING_PROCESS_IMPL_DEF(W)
```

これは `tlhelp32.h` の設計が古くて使いにくいので、ナウいヤングにバカウケの古臭い関数型プログラミング風に Provider/Consumer デザインとしてラップしたもの。

そんでさらに Multibyte と Unicode で同じコードを二度書きたくないからプリプロセッサ使ってジェネリック！テンプレートメタプログラミング！みたいな邪悪マクロによって実装している。

本当はこういうのインテリセンスや静的解析ツールそしてなにより C++ のテンプレート使って書くか C 縛りするにしても何らかのテンプレートエンジン使ってコード生成すべきなのは重々承知であるけど俺こういうの大好きなのよ。
つか C++ テンプレートだってデバッガーのデバッグポイントが壊滅するやろがい、いやちょっとこのへんうろ覚え、なんせ俺 `printf` デバッグおじさんだし(最低野郎)。

C++ 好きな人が俺の書く邪悪マクロを読んだら殺意沸くと思うが Unix C がちょっと書けるだけのおじさんなので勘弁してほしい、つうか最初期の C++ だってスッポスッポ先生はマクロとダンスってたはずである。

つーか `tchar.h` そのものがプリプロセッサ使った邪悪なテンプレートメタプログラミングであって、インテリセンスでマクロをインライン展開しないと引数も見れないんだから、俺だけやっちゃダメな道理がねえよなぁ！？
やはり Win32 API 自体滅ぼさねばならぬ、その結果が C++/WinRT における `winrt::hstring` つまり Unicode への一本化ですって？あっ C から使えないので結構です。

ただしキャプチャとの併用は不可、これはキャプチャ使うとラムダ式は無名関数のポインタではなくキャプチャした変数をメンバに持つ無名クラスのメソッドポインタになるので互換性が無いからしゃーない。

どうせ C++ は ABI なんて無いも同然だし(偏見)今からでも仕様変えてくれねえかな。

ただ上記のコードのままだと Multibyte 系の構造体や関数に末尾 A 存在しないのでビルド通らんのだ、まさか命名規則に沿ってない API があるとは思わず 15 分ほど悩んでしまった。

ちなみにこれら `tlhelp32.h` の関数は今となっては非推奨だから、正気なプログラマであれば使わないはずなので本当にどうでもいい話である。

## `psapi.h` によるプロセス名の列挙

ただし `EnumProcess` が返すのは プロセス ID だけなのでプロセス名まで調べたい今回のケースではさらに以下の処理も必要と MSDN の
[この記事](https://learn.microsoft.com/ja-jp/windows/win32/psapi/enumerating-all-processes)
に書かれている。

* `OpenProcess` でプロセスハンドルを取得
* `EnumProcessModules` あるいは `EnumProcessModulesEx` でモジュールハンドルを取得
* `GetModuleBaseName` でモジュール名を取得

うーんめんどくせえ、そんで `EnumProcess` と `EnumProcessModules` スナップショット取らないから全てのプロセス ID を格納したけりゃできるだけ大きな配列をご用意くださいってのがもうデザインとして終わっとる。
それってスナップショットのコストをこっちに押し付けただけじゃねーかこのカスゥ！

それにそもそも `tlhelp32.h` の置き換えなら .exe 名だけでよくて .dll 名までは不要だし `EnumProcessModules` 呼ぶ必要ねえなこれ。
`OpenProcess` して `QueryFullProcessImageName` すりゃいいだけじゃねえのこれ、余計な記述まで入れるからそこらじゅうのブログで自分で考える頭を持たないコピペマンが無駄な処理までそのまんまコピーしてて笑う。

そんで 32bit プロセスから 64bit プロセスを EnumProcessModules` しようとするとビット数の壁で `ERROR_PARTIAL_COPY` 返ってくる問題も踏むじゃんこれ。
その結果 32bit プロセスからは 64bit プロセスのリストは取得できないなんていう間違ったいかがでしたか記事を鵜呑みにしたおバカ AI が誇らしげに嘘を真実かのように答えるのである。
32bit プロセスからは 64bit プロセスがロードした DLL までは取得できないの間違いやでそれ、名前だけなら問題ないです。

ということで `tlhelp32.h` でなく `psapi.h`　使って上のコードを書き直した結果がこちら、クソみてえに行数増えた。

```c
#include <windows.h>
#include <psapi.h>
#include <tchar.h>

#define CONCAT(a,b)	a##b

#define PAGESIZE (4096)

#define RUNNING_PROCESS_PROTO_DECL(suffix, cstrtype) \
typedef PROC(*CONCAT(RunningProcessIdConsumer,suffix))(PROC, PVOID, DWORD, LSTATUS*); \
LSTATUS CONCAT(RunningProcessIdProvider,suffix)(PVOID, CONCAT(RunningProcessIdConsumer,suffix)); \
typedef PROC(*CONCAT(RunningProcessHandleConsumer,suffix))(PROC, PVOID, DWORD, HANDLE, LSTATUS*); \
LSTATUS CONCAT(RunningProcessHandleProvider,suffix)(PVOID, CONCAT(RunningProcessHandleConsumer,suffix)); \
typedef PROC(*CONCAT(RunningProcessImageFileNameConsumer,suffix))(PROC, PVOID, DWORD, HANDLE, cstrtype, LSTATUS*); \
LSTATUS CONCAT(RunningProcessImageFileNameProvider,suffix)(PVOID, CONCAT(RunningProcessImageFileNameConsumer,suffix));

#if defined(UNICODE)
#define RunningProcessIdConsumer RunningProcessIdConsumerW
#define RunningProcessIdProvider RunningProcessIdProviderW
#define RunningProcessHandleConsumer RunningProcessHandleConsumerW
#define RunningProcessHandleProvider RunningProcessHandleProviderW
#define RunningProcessImageFileNameConsumer RunningProcessImageFileNameConsumerW
#define RunningProcessImageFileNameProvider RunningProcessImageFileNameProviderW
#else
#define RunningProcessIdConsumer RunningProcessIdConsumerA
#define RunningProcessIdProvider RunningProcessIdProviderA
#define RunningProcessHandleConsumer RunningProcessHandleConsumerA
#define RunningProcessHandleProvider RunningProcessHandleProviderA
#define RunningProcessImageFileNameConsumer RunningProcessImageFileNameConsumerA
#define RunningProcessImageFileNameProvider RunningProcessImageFileNameProviderA
#endif

RUNNING_PROCESS_PROTO_DECL(A, PCSTR)
RUNNING_PROCESS_PROTO_DECL(W, PCWSTR)

#define RUNNING_PROCESS_IMPL_DEF(suffix, strtype, cstrtype) \
LSTATUS \
CONCAT(RunningProcessIdProvider,suffix)(PVOID param, CONCAT(RunningProcessIdConsumer,suffix) consume) \
{ \
	LSTATUS ret; \
	HANDLE heap; \
	PBYTE processId; \
	SIZE_T processIdAllocated; \
	DWORD processIdSize; \
	PDWORD q, p; \
	LPVOID tmp; \
 \
	heap = GetProcessHeap(); \
	if (heap == NULL) \
		return GetLastError(); \
	processIdAllocated = PAGESIZE; \
	processId = (PBYTE)HeapAlloc(heap, 0, processIdAllocated); \
	if (processId == NULL) \
		return GetLastError(); \
	ret = ERROR_SUCCESS; \
	for (;;) { \
		if (EnumProcesses((DWORD *)processId, (DWORD)processIdAllocated, &processIdSize) == FALSE) { \
			ret = GetLastError(); \
			break; \
		} \
		if ((SIZE_T)processIdSize <= processIdAllocated) { \
			q = (PDWORD)(processId + processIdSize); \
			for (p = (PDWORD)processId; p < q; ++p) { \
				consume = (CONCAT(RunningProcessIdConsumer,suffix))(*consume)((PROC)consume, param, *p, &ret); \
				if (consume == NULL) \
					break; \
			} \
			break; \
		} \
		if (processIdAllocated >= SIZE_MAX / 2) { \
			ret = ERROR_ARITHMETIC_OVERFLOW; \
			break; \
		} \
		processIdAllocated *= 2; \
		tmp = HeapReAlloc(heap, 0, processId, processIdAllocated); \
		if (tmp == NULL) { \
			ret = GetLastError(); \
			break; \
		} \
		processId = (PBYTE)tmp; \
	} \
	HeapFree(heap, 0, processId); \
	return ret; \
} \
 \
struct CONCAT(RunningProcessHandleDelegateCapture,suffix) { \
	PVOID param; \
	CONCAT(RunningProcessHandleConsumer,suffix)* consume; \
}; \
 \
static PROC \
CONCAT(RunningProcessHandleDelegate,suffix)(PROC self, PVOID param, DWORD processId, LSTATUS* result) \
{ \
	struct CONCAT(RunningProcessHandleDelegateCapture,suffix)* capture; \
	HANDLE processHandle; \
	DWORD ret; \
 \
	capture = (struct CONCAT(RunningProcessHandleDelegateCapture,suffix)*)param; \
	processHandle = OpenProcess(PROCESS_QUERY_INFORMATION|PROCESS_VM_READ, FALSE, processId); \
	if (processHandle == NULL) { \
		ret = GetLastError(); \
		switch (ret) { \
		case ERROR_ACCESS_DENIED: /* process is Client/Server Runtime Subsystem(CSRSS) */ \
		case ERROR_INVALID_PARAMETER: /* process is SystemIdleProcess(id=0) */ \
			break; \
		default: \
			*result = (LSTATUS)ret; \
			self = NULL; \
		} \
	} else { \
		*capture->consume = (CONCAT(RunningProcessHandleConsumer,suffix))(*(*capture->consume))((PROC)*capture->consume, capture->param, processId, processHandle, result); \
		if (*capture->consume == NULL) \
			self = NULL; \
	} \
	return self; \
} \
 \
LSTATUS \
CONCAT(RunningProcessHandleProvider,suffix)(PVOID param, CONCAT(RunningProcessHandleConsumer,suffix) consume) \
{ \
	struct CONCAT(RunningProcessHandleDelegateCapture,suffix) capture = { \
		param, \
		&consume \
	}; \
	return CONCAT(RunningProcessIdProvider,suffix)(&capture, &CONCAT(RunningProcessHandleDelegate,suffix)); \
} \
 \
struct CONCAT(RunningProcessImageFileNameDelegateCapture,suffix) { \
	PVOID param; \
	CONCAT(RunningProcessImageFileNameConsumer,suffix)* consume; \
}; \
 \
static PROC \
CONCAT(RunningProcessImageFileNameDelegate,suffix)(PROC self, PVOID param, DWORD processId, HANDLE processHandle, LSTATUS* result) \
{ \
	DWORD ret; \
	struct CONCAT(RunningProcessImageFileNameDelegateCapture,suffix)* capture; \
	HANDLE heap; \
	strtype path; \
	SIZE_T pathAllocated; \
	DWORD pathSize; \
	LPVOID tmp; \
 \
	ret = ERROR_SUCCESS; \
	capture = (struct CONCAT(RunningProcessImageFileNameDelegateCapture,suffix)*)param; \
	heap = GetProcessHeap(); \
	if (heap == NULL) { \
		ret = GetLastError(); \
	} else { \
		pathAllocated = PAGESIZE; \
		path = (strtype)HeapAlloc(heap, 0, pathAllocated); \
		if (path == NULL) {\
			ret = GetLastError(); \
		} else { \
			for (;;) { \
				pathSize = (DWORD)(pathAllocated / sizeof(*path)); \
				if (CONCAT(QueryFullProcessImageName,suffix)(processHandle, 0, path, &pathSize) == FALSE) { \
					ret = GetLastError(); \
					if (ret != ERROR_INSUFFICIENT_BUFFER) \
						break; \
					ret = ERROR_SUCCESS; \

					*capture->consume = (CONCAT(RunningProcessImageFileNameConsumer,suffix))(*(*capture->consume))((PROC)*capture->consume, capture->param, processId, processHandle, (cstrtype)path, result); \

					break; \
				} \
				if (pathAllocated >= SIZE_MAX / 2) { \
					ret = ERROR_ARITHMETIC_OVERFLOW; \

				pathAllocated *= 2; \
				tmp = HeapReAlloc(heap, 0, path, pathAllocated); \
				if (tmp == NULL) { \
					ret = GetLastError(); \
					break; \
				} \
				path = (strtype)tmp; \
			} \
			HeapFree(heap, 0, path); \
		} \
	} \
	if (ret != ERROR_SUCCESS) { \
		*result = ret; \
		self = NULL; \
	} \
	return self; \
} \

LSTATUS \
CONCAT(RunningProcessImageFileNameProvider,suffix)(PVOID param, CONCAT(RunningProcessImageFileNameConsumer,suffix) consume) \
{ \
	struct CONCAT(RunningProcessImageFileNameDelegateCapture,suffix) capture = { \
		param, \
		&consume \
	}; \

RUNNING_PROCESS_IMPL_DEF(W, PWSTR, PCWSTR)
```

## 今回もまた無駄なコードであった

そんでここまで実装してようやく気づいたんだけど、そもそも CtHelper.exe が PatchMix DSP プロセスが実行されてるか調べたければ、ウインドウタイトルで `FindWindow` すりゃいいだけなんよな。

なんでプロセス一覧から苦労して調べる必要があるねんこのカスボケタニシ！

まぁこれはレジストリから実行ファイル名を取得してるので、ウィンドタイトルはプログラムにハードコードしたくないからと思われる。
