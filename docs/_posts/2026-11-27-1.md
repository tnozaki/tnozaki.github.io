先日の radixsort(3) の件があったので他のコードも監査したのだが、qsort(3) も再帰を使っとる、まぁこっちはスタックの消費量は大きくないし再帰の深さも部分配列の小さい方から処理することで size_t の bit 幅で納まるはずだからスタックオーバーフローは起こすのは無理っぽいので修正せんでええかな、よう知らんけど。
glibc2 の　qsort(3) は再帰せずに実装されてるっぽいが真面目に読む気力が無いので間違ってたらバーカバーカとでも罵ってくれ、どうせ俺には届かないので好きにしろ。

余談だけど qsort(3) は radixsort(3) と違いソート対象が不透明オブジェクトだから数値も文字列もソート可能なのだけど、リスト構造のソートまでは考慮されてないのがアレ。

なのでタイプジェネリックなソート実装欲しいねってことで、いつもの邪悪なマクロで目が腐るコードを書いてみた。

```c++
#include <sys/cdefs.h>
#ifndef __arraycount
#define __arraycount(arr)	(sizeof(arr)/sizeof(arr[0]))
#endif
#include <sys/queue.h>
#include <assert.h>
#include <errno.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

#define QUICK_SORT(name, type, next, prev, cmp, swap)		\
int								\
name##_quick_sort(type l, type r)				\
{								\
	struct stack {						\
		type l;						\
		type r;						\
	} *s, *sp, *tmp;					\
	size_t stack_capacity = 64, len;			\
	type part;						\
	type p;							\
	int ret = 0;						\
								\
	assert(l != NULL);					\
	assert(r != NULL);					\
	if (l == r)						\
		return 0;					\
	s = malloc(stack_capacity * sizeof(*s));		\
	if (s == NULL)						\
		return -1;					\
	sp = s;							\
	sp->l = l;						\
	sp->r = r;						\
	++sp;							\
	while (sp > s) {					\
		--sp;						\
		l = sp->l;					\
		r = sp->r;					\
		part = prev(l);					\
		for (p = l; p != r; p = next(p)) {		\
			if (cmp(p, r) <= 0) {			\
				part = (part == NULL)		\
				    ? l : next(part);		\
				swap(part, p);			\
			}					\
		}						\
		part = (part == NULL) ? l : next(part);		\
		swap(part, r);					\
		if (part != l && prev(part) != l) {		\
			len = sp - s;				\
			if (len == stack_capacity) {		\
				if (stack_capacity >= SIZE_MAX	\
				    / sizeof(*s) / 2) {		\
					errno = EOVERFLOW;	\
					ret = -1;		\
					break;			\
				}				\
				stack_capacity *= 2;		\
				tmp = realloc(s, stack_capacity	\
				    * sizeof(*s));		\
				if (tmp == NULL) {		\
					ret = -1;		\
					break;			\
				}				\
				sp = tmp + len;			\
				s = tmp;			\
			}					\
			sp->l = l;				\
			sp->r = prev(part);			\
			++sp;					\
		}						\
		if (part != r && next(part) != r) {		\
			len = sp - s;				\
			if (len == stack_capacity) {		\
				if (stack_capacity >= SIZE_MAX	\
				    / sizeof(*s) / 2) {		\
					errno = EOVERFLOW;	\
					ret = -1;		\
					break;			\
				}				\
				stack_capacity *= 2;		\
				tmp = realloc(s, stack_capacity	\
				    * sizeof(*s));		\
				if (tmp == NULL) {		\
					ret = -1;		\
					break;			\
				}				\
				sp = tmp + len;			\
				s = tmp;			\
			}					\
			sp->l = l;				\
			sp->l = next(part);			\
			sp->r = r;				\
			++sp;					\
		}						\
	}							\
	free(s);						\
	return ret;						\
}

struct Node {
	struct Node *next, *prev;
	int value;
};

static inline struct Node *
Node_next(struct Node *p)
{
	assert(p != NULL);
	return p->next;
}

static inline struct Node *
Node_prev(struct Node *p)
{
	assert(p != NULL);
	return p->prev;
}

static inline int
Node_cmp(struct Node *a, struct Node *b)
{
	assert(a != NULL);
	assert(b != NULL);
	return a->value - b->value;
}

static inline void
Node_swap(struct Node *a, struct Node *b)
{
	int tmp;
	assert(a != NULL);
	assert(b != NULL);
	tmp = a->value;
	a->value = b->value;
	b->value = tmp;
}

QUICK_SORT(Node, struct Node *, Node_next, Node_prev, Node_cmp, Node_swap)

static inline int *
int_next(int *p)
{
	assert(p != NULL);
	return p + 1;
}

static inline int *
int_prev(int *p)
{
	assert(p != NULL);
	return p - 1;
}

static inline int
int_cmp(int *a, int *b)
{
	assert(a != NULL);
	assert(b != NULL);
	return *a - *b;
}

static inline void
int_swap(int *a, int *b)
{
	int tmp;
	assert(a != NULL);
	assert(b != NULL);
	tmp = *a;
	*a = *b;
	*b = tmp;
}

QUICK_SORT(int, int *, int_next, int_prev, int_cmp, int_swap)

int
main(void)
{
	int arr[] = { 35, 12, 43, 8, 51, 27, 19, 4 };
	size_t last = __arraycount(arr) - 1, i;
	struct Node *head = NULL, *tail = NULL, *p;

	/* int */
	int_quick_sort(arr, arr + last);
	for (i = 0; i <= last; ++i)
		printf("%d ", arr[i]);
	printf("\n");

	/* Node */
	for (i = 0; i <= last; ++i) {
		p = malloc(sizeof(*p));
		if (p == NULL)
			abort();
		p->next = NULL;
		p->prev = tail;
		p->value = arr[i];
		if (tail != NULL)
			tail->next = p;
		else
			head = p;
		tail = p;
	}
	Node_quick_sort(head, tail);
	for (p = head; p != NULL; p = p->next)
		printf("%d ", p->value);
	printf("\n");

	return EXIT_SUCCESS;
}
```

要改良点としては、配列なら左右の部分配列の長さが即座に計算できるので struct stack の動的アロケーションを不要にできるのは上でも書いた通りだが、リストだとループして数え上げないとならんからそこは未対応。

あとソート対象が短い場合、挿入ソートに切り替えた方が高速になるそうだが同じ理由で未対応、今の *BSD の qsort(3) は配列の長さが 7 未満になると挿入ソートに切り替えとる。
